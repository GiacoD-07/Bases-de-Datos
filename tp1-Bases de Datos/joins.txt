INNER JOIN (MySQL / MariaDB)
SELECT e.nombre AS evento, t.nombre AS tipo_evento
FROM evento e
INNER JOIN tipo_evento t ON e.id_tipo = t.id_tipo;
SELECT e.nombre AS evento, s.nombre AS sponsor
FROM evento e
INNER JOIN sponsor s ON e.id_sponsor = s.id_sponsor;

LEFT JOIN
SELECT h.nombre AS hotel, ha.tipo AS habitacion
FROM hotel h
LEFT JOIN habitacion ha ON h.id_hotel = ha.id_hotel;
SELECT l.nombre AS lugar, a.nombre AS actividad
FROM lugar_turistico l
LEFT JOIN lugar_actividad la ON l.id_lugar = la.id_lugar
LEFT JOIN actividad a ON la.id_actividad = a.id_actividad;

RIGHT JOIN
SELECT h.nombre AS hotel, ha.tipo AS habitacion
FROM habitacion ha
RIGHT JOIN hotel h ON ha.id_hotel = h.id_hotel;
SELECT e.nombre AS evento, s.nombre AS sponsor
FROM evento e
RIGHT JOIN sponsor s ON e.id_sponsor = s.id_sponsor;

CROSS JOIN
SELECT h.nombre AS hotel, s.nombre AS servicio
FROM hotel h
CROSS JOIN servicio s;
SELECT t.nombre AS tipo_evento, s.nombre AS sponsor
FROM tipo_evento t
CROSS JOIN sponsor s;
