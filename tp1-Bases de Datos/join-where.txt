INNER JOIN + WHERE

👉 Filtra sobre coincidencias

SELECT * 
FROM evento 
INNER JOIN tipo_evento 
ON evento.id_tipo = tipo_evento.id_tipo
WHERE evento.precio > 2000;
SELECT * 
FROM evento 
INNER JOIN sponsor 
ON evento.id_sponsor = sponsor.id_sponsor
WHERE sponsor.nombre = 'YPF';
🔥 🔗 LEFT JOIN + WHERE

👉 Filtrar después del JOIN

SELECT * 
FROM hotel 
LEFT JOIN habitacion 
ON hotel.id_hotel = habitacion.id_hotel
WHERE habitacion.precio > 15000;
SELECT * 
FROM lugar_turistico 
LEFT JOIN lugar_actividad 
ON lugar_turistico.id_lugar = lugar_actividad.id_lugar
WHERE lugar_turistico.nombre LIKE '%Parque%';
🔥 🔗 LEFT JOIN + WHERE IS NULL (clave)
SELECT * 
FROM hotel 
LEFT JOIN habitacion 
ON hotel.id_hotel = habitacion.id_hotel
WHERE habitacion.id_hotel IS NULL;
SELECT * 
FROM evento 
LEFT JOIN sponsor 
ON evento.id_sponsor = sponsor.id_sponsor
WHERE sponsor.id_sponsor IS NULL;
🔥 🔗 RIGHT JOIN + WHERE
SELECT * 
FROM habitacion 
RIGHT JOIN hotel 
ON habitacion.id_hotel = hotel.id_hotel
WHERE hotel.nombre LIKE '%Hotel%';
SELECT * 
FROM evento 
RIGHT JOIN sponsor 
ON evento.id_sponsor = sponsor.id_sponsor
WHERE sponsor.nombre = 'Pepsi';
🔥 🔗 CROSS JOIN + WHERE

👉 Esto es interesante: el WHERE “simula” un filtro

SELECT * 
FROM tipo_evento 
CROSS JOIN sponsor
WHERE tipo_evento.id_tipo = sponsor.id_sponsor;
SELECT * 
FROM hotel 
CROSS JOIN servicio
WHERE hotel.id_hotel = servicio.id_servicio;
🔥 🔗 FULL OUTER JOIN (simulado) + WHERE

👉 Este es como el ejemplo de tu profe 👇

SELECT * 
FROM evento 
LEFT JOIN sponsor 
ON evento.id_sponsor = sponsor.id_sponsor

UNION

SELECT * 
FROM evento 
RIGHT JOIN sponsor 
ON evento.id_sponsor = sponsor.id_sponsor
WHERE evento.id_evento IS NULL OR sponsor.id_sponsor IS NULL;
SELECT * 
FROM hotel 
LEFT JOIN habitacion 
ON hotel.id_hotel = habitacion.id_hotel

UNION

SELECT * 
FROM hotel 
RIGHT JOIN habitacion 
ON hotel.id_hotel = habitacion.id_hotel
WHERE hotel.id_hotel IS NULL OR habitacion.id_habitacion IS NULL;
🧠 CLAVE QUE TE VA A TOMAR EL PROFE

👉 Ese patrón:

WHERE tabla1.id IS NULL OR tabla2.id IS NULL

→ “mostrar los que NO coinciden en ambos lados”
